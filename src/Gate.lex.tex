\begin{semanticmodule}
definition{
  name: boolRank,
  parameters: [value : Bool],
  result: Nat,
  body: (if value then 1 else 0)
}

definition{
  name: gateRank,
  parameters: [evidence : Access::Evidence, fastForward : Access::Flag, forceAllowed : Access::Flag, approvalsHave : Nat, approvalsNeeded : Nat, checksHave : Nat, checksNeeded : Nat],
  result: Nat,
  body: (if (approvalsNeeded <=? approvalsHave) then (if (checksNeeded <=? checksHave) then (boolRank (Protect::refAllowed evidence fastForward forceAllowed)) else 0) else 0)
}
\end{semanticmodule}
