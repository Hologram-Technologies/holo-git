theorem{

  proof: reflexivity{}
}

theorem{
  name: revoked_never_counts,
  parameters: [
    signed : Access::Flag,

    required : Access::AccessMode
  ],
  statement: ((Access::authorize (record Access::Evidence {signed := signed, revoked := (ctor Access::Flag.yes), live := live, granted := granted, required := required})) = false),

theorem{
  name: permits_transitive,
  parameters: [high : Access::AccessMode, middle : Access::AccessMode, low : Access::AccessMode],
  statement: ((((Access::permits high middle) && (Access::permits middle low)) = true) -> ((Access::permits high low) = true)),
  proof: linear_arithmetic{definitions: [Access::permits]},
