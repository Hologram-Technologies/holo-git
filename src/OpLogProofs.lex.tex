\begin{lexlean}{OpLogProofs}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\useglossary{prism.arch@1.0.0}
\importmodule{OpLog}
\title{Text application model}
\begin{semanticmodule}
theorem{
  name: before_irreflexive,
  parameters: [clock : Nat, author : Nat, serial : Nat],
  statement: ((OpLog::before clock author serial clock author serial) = false),
  proof: linear_arithmetic{definitions: [OpLog::before]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: before_asymmetric,
  parameters: [clockA : Nat, authorA : Nat, idA : Nat, clockB : Nat, authorB : Nat, idB : Nat],
  statement: (((OpLog::before clockA authorA idA clockB authorB idB) = true) -> ((OpLog::before clockB authorB idB clockA authorA idA) = false)),
  proof: linear_arithmetic{definitions: [OpLog::before]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: before_transitive,
  parameters: [
    clockA : Nat,
    authorA : Nat,
    idA : Nat,
    clockB : Nat,
    authorB : Nat,
    idB : Nat,
    clockC : Nat,
    authorC : Nat,
    idC : Nat
  ],
  statement: ((((OpLog::before clockA authorA idA clockB authorB idB) && (OpLog::before clockB authorB idB clockC authorC idC)) = true) -> ((OpLog::before clockA authorA idA clockC authorC idC) = true)),
  proof: linear_arithmetic{definitions: [OpLog::before]},
  axioms: [Classical.choice, Quot.sound, propext]
}
\end{semanticmodule}
\end{lexlean}
